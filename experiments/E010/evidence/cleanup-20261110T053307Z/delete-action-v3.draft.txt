import os, stat, json, hashlib, subprocess, pathlib, datetime
END=datetime.datetime(2026,11,10,6,3,7,tzinfo=datetime.timezone.utc)
MANIFEST_SHA='1f10be83f06a14c035f702fc8c1e2dee098ecdf70ed153b6a138e429a5041425'
PLAN_SHA='930da573986b737cd6fbc2864cca1bcacf34bc8229da911b6913b8c107a3da4f'
NS='3581e693-8019-4068-a40f-311488fab764'
FLAGS=os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW

def check_time(margin=0):
    remaining=(END-datetime.datetime.now(datetime.timezone.utc)).total_seconds()
    if remaining<=margin: raise RuntimeError('original-hardstop')
    return remaining

def identity(z):
    return dict(device=z.st_dev,inode=z.st_ino,ownerUID=z.st_uid,mode=stat.S_IMODE(z.st_mode),type=stat.S_IFMT(z.st_mode))

def parts(relative):
    value=pathlib.PurePosixPath(relative)
    if not relative or value.is_absolute() or '..' in value.parts or value.as_posix()!=relative: raise RuntimeError('unsafe-relative')
    return value.parts

def checked_fstat(fd):
    check_time(); return identity(os.fstat(fd))

def open_root(root):
    check_time()
    # Lexical exact path only; never resolve() or traverse a link.
    wanted=pathlib.Path.home()/'Library/Application Support/RivetExperiments'/('rivet-E010-'+NS)
    if not root.is_absolute() or str(root)!=str(wanted): raise RuntimeError('root-path-change')
    check_time();fd=os.open('/',FLAGS); chain={}
    try:
        chain['/']=checked_fstat(fd);current=pathlib.PurePosixPath('/')
        for part in root.parts[1:]:
            check_time();child=os.open(part,FLAGS,dir_fd=fd)
            try: value=checked_fstat(child)
            except BaseException: os.close(child);raise
            os.close(fd);fd=child;current=current/part;chain[str(current)]=value
        return fd,chain
    except BaseException: os.close(fd);raise

def load_relative(rootfd,relative):
    # Only before the first full scan; all descriptors close before lsof.
    components=parts(relative);check_time();fd=os.dup(rootfd)
    try:
        for part in components[:-1]:
            check_time();child=os.open(part,FLAGS,dir_fd=fd)
            try:
                value=checked_fstat(child)
                if value['ownerUID']!=os.getuid():raise RuntimeError('private-parent-owner')
            except BaseException:os.close(child);raise
            os.close(fd);fd=child
        check_time();leaf=os.open(components[-1],os.O_RDONLY|os.O_NOFOLLOW,dir_fd=fd)
        try:
            info=checked_fstat(leaf)
            if info['ownerUID']!=os.getuid() or info['type']!=stat.S_IFREG:raise RuntimeError('private-file-owner-type')
            chunks=[]
            while True:
                check_time();chunk=os.read(leaf,65536)
                if not chunk:break
                chunks.append(chunk)
            return b''.join(chunks)
        finally:os.close(leaf)
    finally:os.close(fd)

def check_binding(root,rootfd,chain,manifest):
    check_time()
    if str(root)!=manifest['root'] or checked_fstat(rootfd)!=manifest['rootIdentity']:raise RuntimeError('root-change')
    if manifest['rootIdentity']['ownerUID']!=os.getuid():raise RuntimeError('root-owner')
    for path,wanted in manifest['parents'].items():
        check_time()
        if chain.get(path)!=wanted or wanted['type']!=stat.S_IFDIR:raise RuntimeError('parent-change')

def scan(rootfd,expected,rootidentity):
    result={}
    def visit(fd,relative=''):
        check_time()
        with os.scandir(fd) as it:
            for entry in it:
                check_time();rel=(relative+'/' if relative else '')+entry.name
                check_time();z=os.stat(entry.name,dir_fd=fd,follow_symlinks=False);value=identity(z)
                if value!=expected.get(rel) or z.st_dev!=rootidentity['device'] or z.st_uid!=rootidentity['ownerUID']:raise RuntimeError('node-change')
                result[rel]=value
                if stat.S_ISDIR(z.st_mode):
                    check_time();child=os.open(entry.name,FLAGS,dir_fd=fd)
                    try:
                        if checked_fstat(child)!=value:raise RuntimeError('directory-swap')
                        visit(child,rel)
                    finally:os.close(child)
    visit(rootfd);return result

def execute(root,manifest_relative,private_plan_relative):

    try:
        check_time(2);root=pathlib.Path(root)

                        except BaseException:os.close(child);raise

    except (Exception,KeyboardInterrupt) as error:
